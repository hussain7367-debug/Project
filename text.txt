    // --- THREE.JS ENGINE SETUP --- //
    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();
    scene.fog = new THREE.FogExp2(0x030712, 0.025);

    const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 100);
    camera.position.set(0, 8, 18);

    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: false });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.setClearColor(0x030712, 1);
    container.appendChild(renderer.domElement);

    const controls = new THREE.OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.maxDistance = 35;
    controls.minDistance = 2;
    controls.target.set(0, 0, 0);

    // Lighting Setup
    const ambientLight = new THREE.AmbientLight(0xdcfce7, 0.6);
    scene.add(ambientLight);

    const sunLight = new THREE.DirectionalLight(0xfef08a, 1.8);
    sunLight.position.set(10, 20, 10);
    sunLight.castShadow = true;
    scene.add(sunLight);

    const rimLight = new THREE.DirectionalLight(0x38bdf8, 0.8);
    rimLight.position.set(-10, -10, -10);
    scene.add(rimLight);

    // Interactive Raycaster for Picking
    const raycaster = new THREE.Raycaster();
    const mouse = new THREE.Vector2();

    // Simulation State
    const simState = {
      speed: 1.0,
      lightIntensity: 1.0,
      paused: false,
      selectedComponent: 'chloroplast'
    };

    // --- SCENE MESH GENERATION --- //
    const chloroplastGroup = new THREE.Group();
    scene.add(chloroplastGroup);

    const interactiveObjects = [];

    // 1. Outer Membrane Shell (Cutaway Ellipsoid)
    const shellGeo = new THREE.SphereGeometry(6, 48, 32);
    // Scale into an elongated chloroplast shape
    shellGeo.scale(1.5, 0.7, 1.0);

    const shellMat = new THREE.MeshPhysicalMaterial({
      color: 0x15803d,
      transparent: true,
      opacity: 0.35,
      roughness: 0.2,
      transmission: 0.6,
      thickness: 1.2,
      side: THREE.DoubleSide
    });

    const outerMembraneMesh = new THREE.Mesh(shellGeo, shellMat);
    outerMembraneMesh.userData = { id: 'outerMembrane' };
    chloroplastGroup.add(outerMembraneMesh);
    interactiveObjects.push(outerMembraneMesh);

    // 2. Thylakoid Grana Stacks & Intergranal Lamellae
    const granaGroup = new THREE.Group();
    chloroplastGroup.add(granaGroup);

    const discGeo = new THREE.CylinderGeometry(0.8, 0.8, 0.16, 24);
    const thylakoidMat = new THREE.MeshStandardMaterial({
      color: 0x16a34a,
      roughness: 0.3,
      metalness: 0.1
    });

    const ps2Mat = new THREE.MeshStandardMaterial({ color: 0x38bdf8, roughness: 0.2 }); // PSII (Cyan)
    const ps1Mat = new THREE.MeshStandardMaterial({ color: 0xf59e0b, roughness: 0.2 }); // PSI (Amber)
    const atpMat = new THREE.MeshStandardMaterial({ color: 0x10b981, roughness: 0.2 }); // ATP Synthase (Emerald)

    const granaPositions = [
      { x: -3.5, z: -1, count: 6 },
      { x: -1.8, z: 1.2, count: 7 },
      { x: 0, z: -1.5, count: 8 },
      { x: 1.8, z: 1.0, count: 6 },
      { x: 3.5, z: -0.8, count: 7 },
      { x: -0.5, z: 2.2, count: 5 }
    ];

    const granaMeshes = [];

    granaPositions.forEach((pos, gIdx) => {
      const stackGroup = new THREE.Group();
      stackGroup.position.set(pos.x, -0.6, pos.z);

      for (let i = 0; i < pos.count; i++) {
        const disc = new THREE.Mesh(discGeo, thylakoidMat);
        disc.position.y = i * 0.22;
        disc.castShadow = true;
        disc.receiveShadow = true;
        disc.userData = { id: 'granum' };
        stackGroup.add(disc);
        interactiveObjects.push(disc);

        // Add embedded proteins on outer thylakoid margins
        if (i % 2 === 0) {
          // PSII
          const ps2 = new THREE.Mesh(new THREE.BoxGeometry(0.2, 0.2, 0.2), ps2Mat);
          ps2.position.set(0.75, i * 0.22, 0);
          ps2.userData = { id: 'ps2' };
          stackGroup.add(ps2);
          interactiveObjects.push(ps2);

          // PSI
          const ps1 = new THREE.Mesh(new THREE.BoxGeometry(0.2, 0.2, 0.2), ps1Mat);
          ps1.position.set(-0.75, i * 0.22, 0);
          ps1.userData = { id: 'ps1' };
          stackGroup.add(ps1);
          interactiveObjects.push(ps1);
        }

        if (i === pos.count - 1) {
          // ATP Synthase on top
          const atpHead = new THREE.Mesh(new THREE.CylinderGeometry(0.15, 0.25, 0.3, 12), atpMat);
          atpHead.position.set(0, i * 0.22 + 0.2, 0);
          atpHead.userData = { id: 'atpSynthase' };
          stackGroup.add(atpHead);
          interactiveObjects.push(atpHead);
        }
      }
      granaGroup.add(stackGroup);
      granaMeshes.push(stackGroup);
    });

    // Connecting Lamellae Bridges
    const lamellaMat = new THREE.MeshStandardMaterial({ color: 0x15803d, roughness: 0.4 });
    for (let i = 0; i < granaPositions.length - 1; i++) {
      const p1 = granaPositions[i];
      const p2 = granaPositions[i + 1];
      const distance = Math.hypot(p2.x - p1.x, p2.z - p1.z);
      const bridgeGeo = new THREE.BoxGeometry(distance, 0.08, 0.3);
      const bridge = new THREE.Mesh(bridgeGeo, lamellaMat);
      
      bridge.position.set((p1.x + p2.x) / 2, 0, (p1.z + p2.z) / 2);
      bridge.rotation.y = Math.atan2(p2.x - p1.x, p2.z - p1.z) + Math.PI / 2;
      bridge.userData = { id: 'granum' };
      granaGroup.add(bridge);
    }

    // --- 4D PARTICLE FLOW SYSTEM (Molecular Dynamics) --- //

    // A. Light Photon Rays (Sunlight to PSII)
    const photonCount = 60;
    const photonGeo = new THREE.BufferGeometry();
    const photonPositions = new Float32Array(photonCount * 3);

    for (let i = 0; i < photonCount; i++) {
      photonPositions[i * 3] = (Math.random() - 0.5) * 12;
      photonPositions[i * 3 + 1] = 8 + Math.random() * 4;
      photonPositions[i * 3 + 2] = (Math.random() - 0.5) * 8;
    }
    photonGeo.setAttribute('position', new THREE.BufferAttribute(photonPositions, 3));

    const photonMat = new THREE.PointsMaterial({
      color: 0xfef08a,
      size: 0.25,
      transparent: true,
      opacity: 0.9,
      blending: THREE.AdditiveBlending
    });
    const photonParticles = new THREE.Points(photonGeo, photonMat);
    scene.add(photonParticles);

    // B. Electron Transport Chain Particles (e- flowing across grana)
    const eCount = 40;
    const eGeo = new THREE.BufferGeometry();
    const ePositions = new Float32Array(eCount * 3);
    for (let i = 0; i < eCount * 3; i++) ePositions[i] = 0;
    eGeo.setAttribute('position', new THREE.BufferAttribute(ePositions, 3));

    const eMat = new THREE.PointsMaterial({
      color: 0xfcd34d,
      size: 0.2,
      transparent: true,
      opacity: 1.0,
      blending: THREE.AdditiveBlending
    });
    const electronParticles = new THREE.Points(eGeo, eMat);
    chloroplastGroup.add(electronParticles);

    // C. Proton Gradient Flow (H+ from stroma into lumen and through ATP Synthase)
    const hCount = 80;
    const hGeo = new THREE.BufferGeometry();
    const hPos = new Float32Array(hCount * 3);
    for (let i = 0; i < hCount; i++) {
      hPos[i * 3] = (Math.random() - 0.5) * 8;
      hPos[i * 3 + 1] = (Math.random() - 0.5) * 2;
      hPos[i * 3 + 2] = (Math.random() - 0.5) * 5;
    }
    hGeo.setAttribute('position', new THREE.BufferAttribute(hPos, 3));

    const hMat = new THREE.PointsMaterial({
      color: 0x93c5fd,
      size: 0.18,
      transparent: true,
      opacity: 0.85
    });
    const protonParticles = new THREE.Points(hGeo, hMat);
    chloroplastGroup.add(protonParticles);

    // D. Calvin Cycle Ring Flow (Stroma CO2 -> Glucose)
    const calvinCount = 50;
    const calvinGeo = new THREE.BufferGeometry();
    const calvinPos = new Float32Array(calvinCount * 3);
    calvinGeo.setAttribute('position', new THREE.BufferAttribute(calvinPos, 3));

    const calvinMat = new THREE.PointsMaterial({
      color: 0x34d399,
      size: 0.22,
      transparent: true,
      opacity: 0.9
    });
    const calvinParticles = new THREE.Points(calvinGeo, calvinMat);
    calvinParticles.position.set(0, 0, 0);
    chloroplastGroup.add(calvinParticles);

    // --- ANIMATION & TIME-STEP ENGINE --- //
    let clock = new THREE.Clock();

    function updateSimulation(delta) {
      if (simState.paused) return;

      const timeScale = simState.speed * delta;
      const elapsedTime = clock.getElapsedTime() * simState.speed;

      // 1. Animate Photons
      const pPos = photonParticles.geometry.attributes.position.array;
      for (let i = 0; i < photonCount; i++) {
        pPos[i * 3 + 1] -= 8 * timeScale * simState.lightIntensity;
        if (pPos[i * 3 + 1] < -1) {
          pPos[i * 3 + 1] = 10 + Math.random() * 2;
        }
      }
      photonParticles.geometry.attributes.position.needsUpdate = true;

      // 2. Animate Electrons along Grana Pathways
      const ePos = electronParticles.geometry.attributes.position.array;
      for (let i = 0; i < eCount; i++) {
        const t = (elapsedTime * 1.5 + i * 0.2) % (Math.PI * 2);
        ePos[i * 3] = Math.cos(t * 2) * 3.5;
        ePos[i * 3 + 1] = Math.sin(t * 3) * 0.4;
        ePos[i * 3 + 2] = Math.sin(t * 2) * 2.0;
      }
      electronParticles.geometry.attributes.position.needsUpdate = true;

      // 3. Animate Protons (H+) through ATP Synthase
      const hP = protonParticles.geometry.attributes.position.array;
      for (let i = 0; i < hCount; i++) {
        hP[i * 3 + 1] += 1.2 * timeScale;
        if (hP[i * 3 + 1] > 1.8) hP[i * 3 + 1] = -1.2;
      }
      protonParticles.geometry.attributes.position.needsUpdate = true;

      // 4. Animate Calvin Cycle Orbital Ring
      const cPos = calvinParticles.geometry.attributes.position.array;
      const radius = 4.2;
      for (let i = 0; i < calvinCount; i++) {
        const angle = (i / calvinCount) * Math.PI * 2 + elapsedTime * 0.8;
        cPos[i * 3] = Math.cos(angle) * radius;
        cPos[i * 3 + 1] = Math.sin(angle * 2) * 0.5;
        cPos[i * 3 + 2] = Math.sin(angle) * radius;
      }
      calvinParticles.geometry.attributes.position.needsUpdate = true;

      // 5. Gentle Chloroplast Floating Movement
      chloroplastGroup.rotation.y = Math.sin(elapsedTime * 0.2) * 0.15;
    }

    function animate() {
      requestAnimationFrame(animate);
      const delta = clock.getDelta();
      controls.update();
      updateSimulation(delta);
      renderer.render(scene, camera);
    }

    // --- INTERACTIVE PICKING & UI BINDINGS --- //

    function updateCardInfo(id) {
      const info = componentData[id] || componentData['chloroplast'];
      document.getElementById('card-title').textContent = info.title;
      document.getElementById('card-tag').textContent = info.tag;
      document.getElementById('card-type').textContent = info.type;
      document.getElementById('card-desc').textContent = info.desc;
      document.getElementById('card-stat-1').textContent = info.stat1;
      document.getElementById('card-stat-2').textContent = info.stat2;
      simState.selectedComponent = id;
    }

    function onPointerDown(event) {
      // Avoid picking when interacting with HTML UI overlays
      if (event.clientY > window.innerHeight - 120 || event.clientX < 340 && event.clientY < 320) return;

      mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
      mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;

      raycaster.setFromCamera(mouse, camera);
      const intersects = raycaster.intersectObjects(interactiveObjects);

      if (intersects.length > 0) {
        const hit = intersects[0].object;
        if (hit.userData && hit.userData.id) {
          updateCardInfo(hit.userData.id);
        }
      }
    }

    // Camera Preset Smooth Transitions
    function setCameraPreset(preset) {
      document.querySelectorAll('.cam-preset-btn').forEach(btn => {
        btn.classList.remove('active', 'bg-emerald-500/20', 'text-emerald-300', 'border-emerald-500/40');
        btn.classList.add('bg-slate-800/80', 'text-slate-300', 'border-slate-700/50');
      });

      if (preset === 'overview') {
        document.getElementById('cam-overview').classList.add('active', 'bg-emerald-500/20', 'text-emerald-300', 'border-emerald-500/40');
        animateCameraPosition(0, 8, 18, 0, 0, 0);
        updateCardInfo('chloroplast');
      } else if (preset === 'thylakoid') {
        document.getElementById('cam-thylakoid').classList.add('active', 'bg-emerald-500/20', 'text-emerald-300', 'border-emerald-500/40');
        animateCameraPosition(-1.8, 2.5, 4.5, -1.8, 0, 1.2);
        updateCardInfo('granum');
      } else if (preset === 'stroma') {
        document.getElementById('cam-stroma').classList.add('active', 'bg-emerald-500/20', 'text-emerald-300', 'border-emerald-500/40');
        animateCameraPosition(0, 7, 0.1, 0, 0, 0);
        updateCardInfo('stroma');
      }
    }

    function animateCameraPosition(px, py, pz, tx, ty, tz) {
      const startPos = camera.position.clone();
      const targetPos = new THREE.Vector3(px, py, pz);
      const startTarget = controls.target.clone();
      const endTarget = new THREE.Vector3(tx, ty, tz);

      let progress = 0;
      const interval = setInterval(() => {
        progress += 0.05;
        camera.position.lerpVectors(startPos, targetPos, progress);
        controls.target.lerpVectors(startTarget, endTarget, progress);
        if (progress >= 1) clearInterval(interval);
      }, 16);
    }
